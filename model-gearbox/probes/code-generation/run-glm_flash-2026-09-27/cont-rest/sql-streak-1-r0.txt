```sql
WITH d AS (
    SELECT DISTINCT user_id, login_date
    FROM logins
),
g AS (
    SELECT
        user_id,
        login_date,
        LAG(login_date) OVER (
            PARTITION BY user_id ORDER BY login_date
        ) AS prev_date
    FROM d
),
marks AS (
    SELECT
        user_id,
        login_date,
        CASE
            WHEN prev_date IS NOT NULL
                 AND date(prev_date, '+1 day') = login_date
            THEN 0
            ELSE 1
        END AS is_streak_start
    FROM g
),
streaks AS (
    SELECT
        user_id,
        login_date AS streak_start,
        SUM(is_streak_start) OVER (
            PARTITION BY user_id
            ORDER BY login_date
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) AS grp
    FROM marks
),
streak_lengths AS (
    SELECT
        user_id,
        grp,
        streak_start,
        COUNT(*) AS streak_len
    FROM streaks
    GROUP BY user_id, grp
),
ranked AS (
    SELECT
        user_id,
        streak_len AS longest_streak,
        streak_start,
        ROW_NUMBER() OVER (
            PARTITION BY user_id
            ORDER BY streak_len DESC, streak_start ASC
        ) AS rn
    FROM streak_lengths
)
SELECT
    user_id,
    longest_streak,
    streak_start
FROM ranked
WHERE rn = 1
  AND longest_streak >= 3
ORDER BY longest_streak DESC, user_id ASC;
```

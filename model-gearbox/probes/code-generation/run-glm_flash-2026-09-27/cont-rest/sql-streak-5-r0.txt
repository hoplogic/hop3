```sql
WITH d AS (
    SELECT DISTINCT user_id, login_date
    FROM logins
),
g AS (
    SELECT
        user_id,
        login_date,
        date(
            login_date,
            '-' || (ROW_NUMBER() OVER (
                PARTITION BY user_id
                ORDER BY login_date
            ) - 1) || ' days'
        ) AS grp
    FROM d
),
streaks AS (
    SELECT
        user_id,
        MIN(login_date) AS streak_start,
        COUNT(*) AS longest_streak
    FROM g
    GROUP BY user_id, grp
),
ranked AS (
    SELECT
        user_id,
        longest_streak,
        streak_start,
        ROW_NUMBER() OVER (
            PARTITION BY user_id
            ORDER BY longest_streak DESC, streak_start ASC
        ) AS rn
    FROM streaks
)
SELECT user_id, longest_streak, streak_start
FROM ranked
WHERE rn = 1
  AND longest_streak >= 3
ORDER BY longest_streak DESC, user_id ASC;
```

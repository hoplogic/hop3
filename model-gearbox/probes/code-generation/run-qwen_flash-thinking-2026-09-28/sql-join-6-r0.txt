```sql
WITH region_list AS (
    SELECT region, COUNT(*) AS customer_count
    FROM customers
    GROUP BY region
),
paid_orders AS (
    SELECT 
        c.region,
        COUNT(*) AS paid_order_count,
        SUM(COALESCE(o.amount, 0)) AS paid_amount
    FROM orders o
    INNER JOIN customers c ON o.customer_id = c.id
    WHERE o.status = 'paid'
    GROUP BY c.region
)
SELECT 
    rl.region,
    rl.customer_count,
    COALESCE(po.paid_order_count, 0) AS paid_order_count,
    COALESCE(po.paid_amount, 0) AS paid_amount
FROM region_list rl
LEFT JOIN paid_orders po ON rl.region = po.region
ORDER BY paid_amount DESC, rl.region;
```

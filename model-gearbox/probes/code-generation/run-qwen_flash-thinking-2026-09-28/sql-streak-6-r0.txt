```sql
WITH distinct_logins AS (
    SELECT user_id, login_date
    FROM logins
    GROUP BY user_id, login_date
),
ranked_logins AS (
    SELECT
        user_id,
        login_date,
        julianday(login_date) - ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY login_date) AS group_key
    FROM distinct_logins
),
streaks AS (
    SELECT
        user_id,
        MIN(login_date) AS streak_start,
        COUNT(*) AS streak_length
    FROM ranked_logins
    GROUP BY user_id, group_key
),
user_max_streak AS (
    SELECT
        user_id,
        streak_start,
        streak_length,
        ROW_NUMBER() OVER (
            PARTITION BY user_id 
            ORDER BY streak_length DESC, streak_start ASC
        ) AS rn
    FROM streaks
)
SELECT 
    user_id,
    streak_length AS longest_streak,
    streak_start
FROM user_max_streak
WHERE rn = 1 AND streak_length >= 3
ORDER BY longest_streak DESC, user_id ASC;
```

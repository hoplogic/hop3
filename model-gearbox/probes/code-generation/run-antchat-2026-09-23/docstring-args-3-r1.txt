import ast
import re
import sys

def check_node(node):
    if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
        if node.name.startswith('_'):
            return
        docstring = ast.get_docstring(node)
        params = set()
        all_args = list(node.args.posonlyargs) + list(node.args.args) + list(node.args.kwonlyargs)
        for arg in all_args:
            if arg.arg not in ('self', 'cls'):
                params.add(arg.arg)
        if node.args.vararg is not None and node.args.vararg.arg not in ('self', 'cls'):
            params.add(node.args.vararg.arg)
        if node.args.kwarg is not None and node.args.kwarg.arg not in ('self', 'cls'):
            params.add(node.args.kwarg.arg)
        if docstring is None:
            print(str(node.lineno) + ": function " + node.name + " has no docstring")
            sys.exit(1)
        else:
            for param in params:
                pattern = r'(?<!\w)' + re.escape(param) + r'(?!\w)'
                if not re.search(pattern, docstring):
                    print(str(node.lineno) + ": function " + node.name + " docstring misses parameter " + param)
                    sys.exit(1)
    elif isinstance(node, ast.ClassDef):
        for child in ast.iter_child_nodes(node):
            check_node(child)
    else:
        for child in ast.iter_child_nodes(node):
            check_node(child)

```python
import ast
import sys

MAX_LINES = 40


def get_function_name(node):
    return node.name


def check_function_length(source, filepath):
    try:
        tree = ast.parse(source)
    except SyntaxError as e:
        print("{}: parse error {}".format(filepath, e), file=sys.stderr)
        sys.exit(1)

    violations = []

    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            start_line = node.lineno
            end_line = node.end_lineno
            length = end_line - start_line + 1
            if length > MAX_LINES:
                msg = "{}:{} function '{}' is {} lines long (max {})".format(
                    filepath, start_line, get_function_name(node), length, MAX_LINES
                )
                violations.append(msg)

    if violations:
        for v in violations:
            print(v)
        sys.exit(1)
    else:
        sys.exit(0)


def main():
    filepath = sys.argv[1]
    with open(filepath, 'r', encoding='utf-8') as f:
        source = f.read()
    check_function_length(source, filepath)


main()
```

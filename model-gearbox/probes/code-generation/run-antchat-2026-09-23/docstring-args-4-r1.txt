```python
import ast
import re
import sys


tree = ast.parse(open(sys.argv[1], encoding="utf-8").read())
violations = []
for node in ast.walk(tree):
    if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
        if node.name.startswith("_"):
            continue
        docstring = ast.get_docstring(node)
        if docstring is None or len(docstring.strip()) == 0:
            violations.append((node.lineno, "no docstring"))
            continue
        required = []
        for a in node.args.args:
            if a.arg not in ("self", "cls"):
                required.append(a.arg)
        for a in node.args.posonlyargs:
            if a.arg not in ("self", "cls"):
                required.append(a.arg)
        if node.args.vararg is not None:
            required.append(node.args.vararg.arg)
        for a in node.args.kwonlyargs:
            if a.arg not in ("self", "cls"):
                required.append(a.arg)
        if node.args.kwarg is not None:
            required.append(node.args.kwarg.arg)
        for name in required:
            escaped = re.escape(name)
            pattern = r"(?<![a-zA-Z0-9_])" + escaped + r"(?![a-zA-Z0-9_])"
            if re.search(pattern, docstring) is None:
                violations.append((node.lineno, "missing docstring argument: %s" % name))
if violations:
    for lineno, reason in violations:
        print("Line %d: %s" % (lineno, reason), file=sys.stderr)
    sys.exit(1)
```

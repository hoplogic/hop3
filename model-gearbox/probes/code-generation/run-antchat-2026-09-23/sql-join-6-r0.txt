```sql
WITH paid_orders AS (
    SELECT o.customer_id, o.amount
    FROM orders o
    INNER JOIN customers c ON o.customer_id = c.id
    WHERE o.status = 'paid'
)
SELECT 
    c.region,
    COUNT(DISTINCT c.id) AS customer_count,
    COALESCE(COUNT(po.amount), 0) AS paid_order_count,
    COALESCE(SUM(COALESCE(po.amount, 0)), 0) AS paid_amount
FROM customers c
LEFT JOIN paid_orders po ON c.id = po.customer_id
GROUP BY c.region
ORDER BY paid_amount DESC, c.region ASC
```

```python
import ast
import re
import sys

source = open(sys.argv[1], 'r', encoding='utf-8').read()
tree = ast.parse(source)
lines = source.splitlines()

def extract_params(node):
    args = node.args
    param_names = []
    for arg in args.posonlyargs:
        param_names.append(arg.arg)
    for arg in args.args:
        param_names.append(arg.arg)
    if args.vararg:
        param_names.append(args.vararg.arg)
    for arg in args.kwonlyargs:
        param_names.append(arg.arg)
    if args.kwarg:
        param_names.append(args.kwarg.arg)
    return param_names

def get_docstring_node(body):
    if not body:
        return None, None
    first = body[0]
    if isinstance(first, ast.Expr) and isinstance(first.value, ast.Constant) and isinstance(first.value.value, str):
        return first.value.value, first.lineno
    return None, None

violations = []

for node in ast.walk(tree):
    check = False
    if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
        if not node.name.startswith('_'):
            check = True
    elif isinstance(node, ast.ClassDef):
        check = False
    else:
        continue

    if not check:
        continue

    params = extract_params(node)
    required = [p for p in params if p not in ('self', 'cls')]

    docstring, doc_line = get_docstring_node(node.body)

    if docstring is None:
        violations.append((node.lineno, f"missing docstring"))
        continue

    for param in required:
        pattern = r'(?<![a-zA-Z0-9_])' + re.escape(param) + r'(?![a-zA-Z0-9_])'
        if not re.search(pattern, docstring):
            violations.append((node.lineno, f"parameter '{param}' not mentioned in docstring"))

if violations:
    for lineno, reason in violations:
        print(f"{lineno}: {reason}")
    sys.exit(1)
else:
    sys.exit(0)
```

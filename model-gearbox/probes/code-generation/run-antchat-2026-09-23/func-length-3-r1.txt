import ast
import sys


def check_functions(tree, source):
    results = []
    lines = source.splitlines()

    def get_end(node):
        return getattr(node, 'end_lineno', node.lineno)

    def func_length(func):
        body = func.body
        if not body:
            return 1
        start = func.lineno
        if func.decorator_list:
            dec_ends = [get_end(d) for d in func.decorator_list]
            start = max(start, max(dec_ends) + 1)
        return get_end(body[-1]) - start + 1

    def collect_funcs(node):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            length = func_length(node)
            if length > 40:
                results.append((node.lineno, length))
            for child in ast.iter_child_nodes(node):
                collect_funcs(child)
        elif isinstance(node, ast.ClassDef):
            for item in node.body:
                if isinstance(item, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    length = func_length(item)
                    if length > 40:
                        results.append((item.lineno, length))
                collect_funcs(item)
        else:
            for child in ast.iter_child_nodes(node):
                collect_funcs(child)

    collect_funcs(tree)
    return results


def main():
    path = sys.argv[1]
    with open(path, 'r', encoding='utf-8') as f:
        source = f.read()
    tree = ast.parse(source)
    violations = check_functions(tree, source)
    if violations:
        for lineno, length in violations:
            print(f"line {lineno}: function is {length} lines long (exceeds 40)")
        sys.exit(1)
    sys.exit(0)


main()

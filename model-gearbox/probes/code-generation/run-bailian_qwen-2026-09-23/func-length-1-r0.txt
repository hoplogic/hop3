import ast
import sys

def check_function_lengths(file_path):
    with open(file_path, 'r', encoding='utf-8') as f:
        content = f.read()
    
    try:
        tree = ast.parse(content)
    except SyntaxError:
        # If syntax error, we just won't find any functions, so compliant
        return []
    
    violations = []
    
    # Stack to keep track of current function context
    def analyze_node(node, parent_func=None):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            # Calculate the length of this function
            start_line = node.lineno
            
            # Get the end line of the function body
            end_line = node.end_lineno
            
            func_length = end_line - start_line + 1
            
            if func_length > 40:
                func_name = node.name
                violations.append(f"Line {start_line}: Function '{func_name}' is too long ({func_length} lines, maximum is 40)")
            
            # Analyze nested functions within this function's body
            for child in ast.iter_child_nodes(node):
                analyze_node(child, node)
        elif isinstance(node, ast.ClassDef):
            # Handle methods inside classes
            for child in ast.iter_child_nodes(node):
                analyze_node(child, parent_func)
        else:
            # For non-function nodes that are not inside a function,
            # we don't need special handling since they're at module level
            for child in ast.iter_child_nodes(node):
                analyze_node(child, parent_func)

    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            start_line = node.lineno
            end_line = node.end_lineno
            func_length = end_line - start_line + 1
            
            if func_length > 40:
                func_name = node.name
                violations.append(f"Line {start_line}: Function '{func_name}' is too long ({func_length} lines, maximum is 40)")
            
            # Check for nested functions inside this one
            for child in ast.iter_child_nodes(node):
                if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    analyze_node(child, node)
        elif isinstance(node, ast.ClassDef):
            # Process methods inside classes
            for child in ast.iter_child_nodes(node):
                if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    start_line = child.lineno
                    end_line = child.end_lineno
                    func_length = end_line - start_line + 1
                    
                    if func_length > 40:
                        func_name = child.name
                        violations.append(f"Line {start_line}: Function '{func_name}' is too long ({func_length} lines, maximum is 40)")
                    
                    # Recursively check for nested functions
                    analyze_node(child, None)

    return violations

file_path = sys.argv[1]
violations = check_function_lengths(file_path)

if violations:
    for violation in violations:
        print(violation)
    sys.exit(1)
else:
    sys.exit(0)

```python
import ast
import sys

def check_function_lengths(file_path):
    with open(file_path, 'r', encoding='utf-8') as f:
        content = f.read()
    
    try:
        tree = ast.parse(content)
    except SyntaxError:
        # If syntax error, we just won't find functions, which is fine
        return []
    
    violations = []
    
    # Walk through all nodes to find function definitions
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            # Calculate function length based on line numbers
            start_line = node.lineno
            end_line = node.end_lineno
            
            # Count the lines of this function (inclusive)
            func_length = end_line - start_line + 1
            
            if func_length > 40:
                violations.append((start_line, f"Function '{node.name}' is too long ({func_length} lines, max 40)"))
    
    return violations

file_path = sys.argv[1]
violations = check_function_lengths(file_path)

if violations:
    for line_num, reason in violations:
        print(f"Line {line_num}: {reason}")
    sys.exit(1)
else:
    sys.exit(0)
```

```sql
WITH unique_logins AS (
    SELECT DISTINCT user_id, login_date
    FROM logins
),
ranked_logins AS (
    SELECT 
        user_id,
        login_date,
        ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY login_date) as rn
    FROM unique_logins
),
grouped_by_streak AS (
    SELECT 
        user_id,
        login_date,
        DATE(login_date, '-' || rn || ' days') as grp
    FROM ranked_logins
),
streaks AS (
    SELECT 
        user_id,
        grp,
        COUNT(*) as streak_length,
        MIN(login_date) as streak_start
    FROM grouped_by_streak
    GROUP BY user_id, grp
),
max_streaks AS (
    SELECT 
        user_id,
        MAX(streak_length) as longest_streak
    FROM streaks
    GROUP BY user_id
    HAVING MAX(streak_length) >= 3
)
SELECT 
    s.user_id,
    ms.longest_streak,
    MIN(s.streak_start) as streak_start
FROM streaks s
JOIN max_streaks ms ON s.user_id = ms.user_id AND s.streak_length = ms.longest_streak
GROUP BY s.user_id, ms.longest_streak
ORDER BY ms.longest_streak DESC, s.user_id ASC;
```

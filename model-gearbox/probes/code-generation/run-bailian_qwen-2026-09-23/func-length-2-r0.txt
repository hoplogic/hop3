```python
import ast
import sys

def check_function_lengths(file_path):
    with open(file_path, 'r', encoding='utf-8') as f:
        content = f.read()
    
    try:
        tree = ast.parse(content)
    except SyntaxError:
        # If syntax error, we just won't find any functions, which is fine
        return []
    
    violations = []
    
    # We'll use a stack to keep track of nested function contexts
    def analyze_node(node, parent_functions=None):
        if parent_functions is None:
            parent_functions = []
        
        current_parent_functions = parent_functions[:]
        
        # Check if this node is a function definition
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            # Calculate function length (from def line to end of function body)
            start_line = node.lineno
            end_line = node.end_lineno
            
            # Function length includes both start and end lines
            func_length = end_line - start_line + 1
            
            if func_length > 40:
                # Get function name for error message
                func_name = node.name
                violations.append(f"Line {start_line}: Function '{func_name}' is too long ({func_length} lines, max 40)")
            
            # Add this function to the parent functions list for nested analysis
            current_parent_functions.append((start_line, end_line, node.name))
        
        # Recursively analyze child nodes
        for child in ast.iter_child_nodes(node):
            analyze_node(child, current_parent_functions)
    
    analyze_node(tree)
    return violations

file_path = sys.argv[1]
violations = check_function_lengths(file_path)

if violations:
    for violation in violations:
        print(violation)
    sys.exit(1)
else:
    sys.exit(0)
```

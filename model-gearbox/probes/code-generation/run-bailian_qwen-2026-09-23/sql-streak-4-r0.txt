```sql
WITH unique_logins AS (
    SELECT DISTINCT user_id, login_date
    FROM logins
),
ranked_logins AS (
    SELECT 
        user_id,
        login_date,
        ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY login_date) as rn
    FROM unique_logins
),
grouped_by_streak AS (
    SELECT 
        user_id,
        login_date,
        DATE(login_date, '-' || rn || ' days') as grp
    FROM ranked_logins
),
streaks AS (
    SELECT 
        user_id,
        grp,
        MIN(login_date) as streak_start,
        COUNT(*) as streak_length
    FROM grouped_by_streak
    GROUP BY user_id, grp
),
max_streaks AS (
    SELECT 
        user_id,
        MAX(streak_length) as longest_streak
    FROM streaks
    WHERE streak_length >= 3
    GROUP BY user_id
),
final_result AS (
    SELECT 
        s.user_id,
        s.streak_length as longest_streak,
        MIN(s.streak_start) as streak_start
    FROM streaks s
    INNER JOIN max_streaks ms ON s.user_id = ms.user_id AND s.streak_length = ms.longest_streak
    WHERE s.streak_length >= 3
    GROUP BY s.user_id, s.streak_length
)
SELECT 
    user_id,
    longest_streak,
    streak_start
FROM final_result
ORDER BY longest_streak DESC, user_id ASC;
```

```sql
WITH daily_logins AS (
  SELECT user_id, login_date
  FROM logins
  GROUP BY user_id, login_date
),
user_streaks AS (
  SELECT 
    user_id,
    login_date,
    -- 判断是否为连续登录的起点：前一天没有登录
    CASE 
      WHEN LAG(login_date) OVER (PARTITION BY user_id ORDER BY login_date) IS NULL 
           OR julianday(login_date) - julianday(LAG(login_date) OVER (PARTITION BY user_id ORDER BY login_date)) != 1
      THEN 1 
      ELSE 0 
    END AS is_streak_start
  FROM daily_logins
),
streak_groups AS (
  SELECT 
    user_id,
    login_date,
    SUM(is_streak_start) AS streak_group_id
  FROM user_streaks
  GROUP BY user_id, login_date
),
streak_lengths AS (
  SELECT 
    user_id,
    MIN(login_date) AS streak_start,
    COUNT(*) AS longest_streak
  FROM streak_groups
  GROUP BY user_id, streak_group_id
),
max_streak_per_user AS (
  SELECT 
    user_id,
    MAX(longest_streak) AS max_streak
  FROM streak_lengths
  GROUP BY user_id
),
ranked_streaks AS (
  SELECT 
    sl.user_id,
    sl.longest_streak,
    sl.streak_start,
    ROW_NUMBER() OVER (
      PARTITION BY sl.user_id 
      ORDER BY sl.longest_streak DESC, sl.streak_start ASC
    ) AS rn
  FROM streak_lengths sl
  JOIN max_streak_per_user ms ON sl.user_id = ms.user_id AND sl.longest_streak = ms.max_streak
)
SELECT 
  user_id,
  longest_streak,
  streak_start
FROM ranked_streaks
WHERE longest_streak >= 3
ORDER BY longest_streak DESC, user_id ASC;
```

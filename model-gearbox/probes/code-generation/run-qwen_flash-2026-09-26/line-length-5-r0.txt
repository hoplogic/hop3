```python
import sys
import re

def check_file_length(filename):
    try:
        with open(filename, 'rt', encoding='utf-8') as file:
            lines = file.readlines()
    except OSError as e:
        sys.stderr.write(f"Error reading file: {e}\n")
        sys.exit(1)

    violations = []
    for i, line in enumerate(lines, start=1):
        # 去掉行尾换行符（包括 \r\n 里的 \r）但保留内容长度计算
        stripped_line = line.rstrip('\r\n')
        if len(stripped_line) > 100:
            violations.append(f"Line {i}: {len(stripped_line)} characters (exceeds 100)")

    if violations:
        for violation in violations:
            sys.stderr.write(violation + '\n')
        sys.exit(1)
    else:
        sys.exit(0)

if __name__ == "__main__":
    if len(sys.argv) < 2:
        sys.stderr.write("Usage: python script.py <filename>\n")
        sys.exit(1)
    
    filename = sys.argv[1]
    check_file_length(filename)
```

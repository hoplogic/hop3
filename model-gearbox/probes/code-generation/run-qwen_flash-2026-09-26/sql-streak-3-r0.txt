```sql
WITH daily_logins AS (
  SELECT user_id, login_date
  FROM logins
  GROUP BY user_id, login_date
),
user_streaks AS (
  SELECT 
    user_id,
    login_date,
    -- 计算当前登录日与前一天是否连续
    CASE 
      WHEN julianday(login_date) - julianday(LAG(login_date) OVER (PARTITION BY user_id ORDER BY login_date)) = 1 
      THEN 0 
      ELSE 1 
    END AS new_streak_flag
  FROM daily_logins
),
streak_groups AS (
  SELECT 
    user_id,
    login_date,
    SUM(new_streak_flag) AS streak_group_id
  FROM user_streaks
  GROUP BY user_id, login_date
),
streak_lengths AS (
  SELECT 
    user_id,
    MIN(login_date) AS streak_start,
    COUNT(*) AS longest_streak
  FROM streak_groups
  GROUP BY user_id, streak_group_id
),
ranked_streaks AS (
  SELECT 
    user_id,
    longest_streak,
    streak_start,
    ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY longest_streak DESC, streak_start ASC) AS rn
  FROM streak_lengths
)
SELECT 
  user_id,
  longest_streak,
  streak_start
FROM ranked_streaks
WHERE longest_streak >= 3
ORDER BY longest_streak DESC, user_id ASC;
```

```python
import sys
import re

def main():
    if len(sys.argv) < 2:
        sys.exit(1)
    
    file_path = sys.argv[1]
    
    try:
        with open(file_path, 'rt', encoding='utf-8') as f:
            lines = f.readlines()
    except Exception as e:
        if isinstance(e, (OSError, ValueError)):
            print(f"Error reading file: {e}", file=sys.stderr)
            sys.exit(1)
        else:
            print(f"Error reading file: {e}", file=sys.stderr)
            sys.exit(1)
    
    # 处理每一行，检查长度
    for i, line in enumerate(lines, start=1):
        # 去掉行尾的换行符（包括 \r\n 中的 \r）
        stripped_line = line.rstrip('\r\n')
        
        # 计算字符数（汉字也算一个字符）
        length = len(stripped_line)
        
        # 检查是否超过 100 字符
        if length > 100:
            print(f"Line {i}: {length} characters (exceeds 100)", file=sys.stderr)
            sys.exit(1)
    
    # 如果没有违规，退出码为 0
    sys.exit(0)

main()
```

```sql
WITH region_customer_counts AS (
    SELECT 
        c.region,
        COUNT(c.id) AS customer_count
    FROM customers c
    GROUP BY c.region
),
paid_orders_summary AS (
    SELECT 
        c.region,
        COUNT(o.id) AS paid_order_count,
        SUM(CASE WHEN o.amount IS NULL THEN 0 ELSE o.amount END) AS paid_amount
    FROM customers c
    LEFT JOIN orders o ON c.id = o.customer_id AND o.status = 'paid'
    GROUP BY c.region
),
combined_results AS (
    SELECT 
        rcc.region,
        rcc.customer_count,
        COALESCE(pos.paid_order_count, 0) AS paid_order_count,
        COALESCE(pos.paid_amount, 0) AS paid_amount
    FROM region_customer_counts rcc
    LEFT JOIN paid_orders_summary pos ON rcc.region = pos.region
)
SELECT 
    region,
    customer_count,
    paid_order_count,
    paid_amount
FROM combined_results
ORDER BY paid_amount DESC, region ASC;
```

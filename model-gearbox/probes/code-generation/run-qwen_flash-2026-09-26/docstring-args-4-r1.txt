def get_parameters(func_node):
    params = []
    for arg in func_node.args.posonlyargs + func_node.args.args:
        if arg.arg not in ('self', 'cls'):
            params.append(arg.arg)
    if func_node.args.vararg and func_node.args.vararg.arg not in ('self', 'cls'):
        params.append(func_node.args.vararg.arg)
    for arg in func_node.args.kwonlyargs:
        if arg.arg not in ('self', 'cls'):
            params.append(arg.arg)
    if func_node.args.kwarg and func_node.args.kwarg.arg not in ('self', 'cls'):
        params.append(func_node.args.kwarg.arg)
    return params

def extract_docstring(node):
    if node.body and isinstance(node.body[0], ast.Expr):
        first = node.body[0]
        if isinstance(first.value, ast.Str):
            return first.value.s
    return None

def is_valid_docstring(docstring, params):
    if not docstring:
        return False
    text = docstring.strip()
    for param in params:
        pattern = r'\b' + re.escape(param) + r'\b'
        if not re.search(pattern, text):
            return False
    return True

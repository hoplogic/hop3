import ast
import re
import json
import sys

def check_function_length(node, filename):
    violations = []
    
    def traverse(node, start_line):
        if isinstance(node, ast.FunctionDef) or isinstance(node, ast.AsyncFunctionDef):
            # 计算函数体行数：从 def 行到 body 最后一行（含）
            # 装饰器不计入，但 def 本身要算
            func_start = node.lineno
            func_end = node.end_lineno
            
            # 检查是否超过 40 行
            if func_end - func_start + 1 > 40:
                violations.append(f"{filename}:{func_start}-{func_end}: 函数长度 {func_end - func_start + 1} 行，超过 40 行限制")
            
            # 递归检查嵌套函数
            for child in ast.iter_child_nodes(node):
                traverse(child, func_start)
        
        elif isinstance(node, ast.ClassDef):
            # 类内方法也要检查
            for child in ast.iter_child_nodes(node):
                traverse(child, node.lineno)
        
        else:
            # 其他节点继续遍历
            for child in ast.iter_child_nodes(node):
                traverse(child, start_line)
    
    traverse(node, 1)
    return violations

def main():
    if len(sys.argv) < 2:
        sys.exit(1)
    
    filepath = sys.argv[1]
    
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            content = f.read()
    except (OSError, UnicodeDecodeError) as e:
        sys.stderr.write(f"无法读取文件: {filepath}\n")
        sys.exit(1)
    
    try:
        tree = ast.parse(content, filename=filepath)
    except SyntaxError as e:
        sys.stderr.write(f"{filepath}:{e.lineno}: 语法错误: {e.msg}\n")
        sys.exit(1)
    
    violations = check_function_length(tree, filepath)
    
    if violations:
        for violation in violations:
            print(violation)
        sys.exit(1)
    else:
        sys.exit(0)

main()

```python
import ast
import re
import json
import sys

def check_function_length(filename):
    with open(filename, 'r', encoding='utf-8') as f:
        content = f.read()

    try:
        tree = ast.parse(content)
    except SyntaxError as e:
        print(f"SyntaxError in {filename}: {e}", file=sys.stderr)
        sys.exit(1)

    violations = []

    def visit_node(node):
        if isinstance(node, ast.FunctionDef) or isinstance(node, ast.AsyncFunctionDef):
            start_line = node.lineno
            end_line = node.end_lineno
            func_lines = end_line - start_line + 1
            if func_lines > 40:
                violations.append(f"Line {start_line}: Function '{node.name}' has {func_lines} lines, exceeds 40-line limit.")
            
            for child in ast.iter_child_nodes(node):
                if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    visit_node(child)

        elif isinstance(node, ast.ClassDef):
            for child in ast.iter_child_nodes(node):
                if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    visit_node(child)

        for child in ast.iter_child_nodes(node):
            visit_node(child)

    visit_node(tree)

    if violations:
        for violation in violations:
            print(violation, file=sys.stderr)
        sys.exit(1)
    else:
        sys.exit(0)

filename = sys.argv[1]
check_function_length(filename)
```

```python
import sys
import re

def check_file_line_length(file_path):
    try:
        with open(file_path, 'rt', encoding='utf-8') as f:
            lines = f.readlines()
    except OSError as e:
        print(f"Error reading file: {e}", file=sys.stderr)
        sys.exit(1)

    violations = []
    for i, line in enumerate(lines, start=1):
        stripped_line = line.rstrip('\r\n')
        if len(stripped_line) > 100:
            violations.append(f"Line {i}: {len(stripped_line)} characters (exceeds 100)")

    if violations:
        for violation in violations:
            print(violation, file=sys.stderr)
        sys.exit(1)
    else:
        sys.exit(0)

if len(sys.argv) < 2:
    print("Usage: python script.py <file_path>", file=sys.stderr)
    sys.exit(1)

file_path = sys.argv[1]
check_file_line_length(file_path)
```

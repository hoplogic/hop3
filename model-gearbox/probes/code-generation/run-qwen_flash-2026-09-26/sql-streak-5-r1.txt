```sql
WITH daily_logins AS (
  SELECT user_id, login_date
  FROM logins
  GROUP BY user_id, login_date
),
ranked_logins AS (
  SELECT 
    user_id,
    login_date,
    ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY login_date) AS rn
  FROM daily_logins
),
streak_groups AS (
  SELECT 
    user_id,
    login_date,
    -- 用 julianday 计算日期差，确保跨月、跨年正确
    julianday(login_date) - rn AS group_key
  FROM ranked_logins
),
streak_stats AS (
  SELECT 
    user_id,
    group_key,
    MIN(login_date) AS streak_start,
    COUNT(*) AS streak_length
  FROM streak_groups
  GROUP BY user_id, group_key
),
max_streaks AS (
  SELECT 
    user_id,
    MAX(streak_length) AS longest_streak,
    MIN(streak_start) AS streak_start
  FROM streak_stats
  GROUP BY user_id
)
SELECT 
  user_id,
  longest_streak,
  streak_start
FROM max_streaks
WHERE longest_streak >= 3
ORDER BY longest_streak DESC, user_id ASC;
```

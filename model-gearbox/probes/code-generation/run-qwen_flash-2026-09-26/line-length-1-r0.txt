```python
import sys
import re

def check_line_length(file_path):
    try:
        with open(file_path, 'r', encoding='utf-8') as file:
            lines = file.readlines()
    except OSError as e:
        sys.stderr.write(f"Error opening file: {e}\n")
        sys.exit(1)

    violations = []
    for i, line in enumerate(lines, start=1):
        # 去掉行尾的换行符和可能的 \r (Windows 换行)
        stripped_line = line.rstrip('\r\n')
        if len(stripped_line) > 100:
            violations.append(f"Line {i}: {len(stripped_line)} characters (exceeds 100)")

    if violations:
        for violation in violations:
            sys.stderr.write(violation + '\n')
        sys.exit(1)
    else:
        sys.exit(0)

if __name__ == "__main__":
    if len(sys.argv) < 2:
        sys.stderr.write("Usage: script.py <file_path>\n")
        sys.exit(1)
    
    file_path = sys.argv[1]
    check_line_length(file_path)
```
